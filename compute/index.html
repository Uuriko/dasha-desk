Local quick start

Three terminals. One model.

git clone https://github.com/Uuriko/dasha-desk.git
cd dasha-desk/compute

# terminal 1
DASHA_API_KEY=consumer-secret DASHA_PROVIDER_KEY=provider-secret npm start

# terminal 2
ollama pull qwen3:8b
DASHA_PROVIDER_KEY=provider-secret DASHA_MODEL_MAP=qwen3-8b=qwen3:8b python3 provider/agent.py

# terminal 3
DASHA_API_KEY=consumer-secret python3 examples/chat.py
